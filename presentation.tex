\section{1. Introduction}

\begin{frame}{Introduction}
\begin{itemize}
    \item Cities are expanding rapidly, and road infrastructure is struggling to keep pace.
    \item Most traffic signals still operate on fixed timers, regardless of real-time traffic conditions.
    \item This project explores how AI can enable intersections to adapt dynamically.
    \item Smarter traffic control reduces commute time, fuel consumption, and emergency delays.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
% Problem Statement
% ---------------------------------------------------------
\section{2. Problem Statement}

\begin{frame}{Problem Statement}
\begin{itemize}
    \item Traditional traffic signals follow pre-set timing cycles.
    \item They cannot adapt to sudden traffic surges or empty lanes.
    \item This results in:
    \begin{itemize}
        \item Unnecessary waiting at red lights
        \item Heavy congestion during peak hours
        \item Delayed emergency vehicle movement
    \end{itemize}
    \item \textbf{Core Issue:} Lack of an intelligent system that dynamically adjusts signal timing based on live road conditions.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
% Objective

\section{3. Objective}

\begin{frame}{Objective}
\textbf{Main Goal:}
\begin{itemize}
    \item Develop an AI-powered adaptive traffic signal control system.
\end{itemize}

\textbf{Specific Objectives:}
\begin{itemize}
    \item Real-time vehicle detection and tracking.
    \item Short-term congestion prediction using sequence models.
    \item Emergency vehicle prioritization.
    \item Illegal parking detection at intersections.
\end{itemize}
\end{frame}

\begin{frame}{Target Outcome}
\textbf{Target Outcome:}
\begin{itemize}
    \item Reduce average waiting time by 30--35\%.
    \item Maintain sub-second response latency.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
% Requirements

\section{5. Requirements}

\begin{frame}{Requirements}
\textbf{Functional Requirements}
\begin{itemize}
    \item Real-time vehicle counting (Cars, Buses, Trucks, Motorcycles).
    \item Lane-based detection using polygon masking.
    \item AI-based dynamic signal switching.
    \item Emergency vehicle green corridor feature.
    \item Illegal parking detection for stationary vehicles.
\end{itemize}

\vspace{0.3cm}

\textbf{Non-Functional Requirements}
\begin{itemize}
    \item Low latency video streaming (< 1 sec delay).
    \item Scalable architecture for multiple intersections.
    \item Reliable logging and fault tolerance.
\end{itemize}
\end{frame}

% ---------------------------------------------------------
% Design
% ---------------------------------------------------------
\section{6. Design}

\begin{frame}{System Architecture (1/2)}
\textbf{1. Perception Layer}
\begin{itemize}
    \item Processes camera feeds.
    \item Applies lane masking.
    \item Uses YOLO-based tracking for vehicle detection.
\end{itemize}

\textbf{2. Decision Layer}
\begin{itemize}
    \item Deep Q-Network for signal decision making.
    \item Bi-LSTM model for congestion prediction.
\end{itemize}
\end{frame}

\begin{frame}{System Architecture (2/2)}
\textbf{3. Communication Layer}
\begin{itemize}
    \item REST APIs for signal control.
    \item WebRTC for live streaming.
\end{itemize}

\textbf{4. Presentation Layer}
\begin{itemize}
    \item Interactive React dashboard.
    \item Real-time metrics and alerts.
\end{itemize}
\end{frame}
